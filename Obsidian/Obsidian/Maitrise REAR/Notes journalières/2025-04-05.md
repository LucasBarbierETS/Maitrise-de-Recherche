##### 🔁 Récurrentes
```dataviewjs
const today = moment(); // Date d'aujourd'hui
const noteDate = moment(dv.current().file.name, "YYYY-MM-DD"); // Date de la note

const tasks = dv.pages().file.tasks
  .where(t => !t.completed && t.recurrence && t.next && moment(t.next).isSame(today, "day"));

dv.taskList(tasks, false);
```
##### ⏳ Planifiées pour cette date
```dataviewjs
const date = dv.current().file.name; // Récupère la date de la note
const tasks = dv.pages().file.tasks
  .where(t => (t.scheduled && t.scheduled.toISODate() === date) || 
              (t.start && t.start.toISODate() === date) || 
              (t.recurrence && t.next && t.next.toISODate() === date)); // Sélectionne les tâches répétitives qui tombent aujourd'hui

dv.taskList(tasks, false);
```
##### 📅 A finir pour cette date
```dataviewjs
const date = dv.current().file.name; // Récupère la date de la note
dv.taskList(dv.pages().file.tasks
  .where(t => t.due && t.due.toISODate() === date), false);
```

##### 🔺En retard
```dataviewjs
const today = moment(); // Date d'aujourd'hui
const noteDate = moment(dv.current().file.name, "YYYY-MM-DD"); // Date de la note

const tasks = dv.pages().file.tasks
  .where(t => !t.completed && t.due && moment(t.due).isBefore(today, "day") && moment(t.due).isSameOrBefore(noteDate, "day"));

dv.taskList(tasks, false);
```
